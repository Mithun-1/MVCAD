cmake_minimum_required(VERSION 3.24)
project(MVCAD VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
qt_standard_project_setup()
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.tools/occt")
find_package(OpenCASCADE 7.9.3 EXACT REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)

add_library(mvcad_core src/core/Network.cpp src/core/PartIO.cpp src/core/Preview.cpp src/core/PointWorkflow.cpp)
target_include_directories(mvcad_core PUBLIC src)
target_link_libraries(mvcad_core PUBLIC Qt6::Core Qt6::Gui)
if(MSVC)
  target_compile_options(mvcad_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mvcad_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(mvcad_cad src/cad/CadModel.cpp)
target_include_directories(mvcad_cad SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(mvcad_cad PUBLIC mvcad_core PRIVATE ${OpenCASCADE_FoundationClasses_LIBRARIES} ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_library(mvcad_document src/core/Document.cpp src/core/DocumentEditing.cpp)
target_link_libraries(mvcad_document PUBLIC mvcad_cad)
add_library(mvcad_builds src/app/GeometryBuildQueue.cpp)
target_link_libraries(mvcad_builds PUBLIC mvcad_document Qt6::Concurrent)
qt_add_executable(MVCAD WIN32 MACOSX_BUNDLE
  src/app/main.cpp src/app/MainWindow.cpp src/app/MainWindowNetwork.cpp src/app/MainWindow.h
  src/app/Viewport.cpp src/app/ViewportSolids.cpp src/app/Viewport.h
  src/app/MainWindowBodies.cpp src/app/MainWindowPoints.cpp src/app/MainWindowVerification.cpp)
target_link_libraries(MVCAD PRIVATE mvcad_builds Qt6::Widgets)
target_compile_definitions(MVCAD PRIVATE MVCAD_VERSION="${PROJECT_VERSION}-dev")
if(MSVC)
  target_compile_options(MVCAD PRIVATE /W4 /permissive- /utf-8)
endif()
set_target_properties(MVCAD PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.mvcad.MVCAD
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})

if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(mvcad_tests tests/CoreTests.cpp)
  target_link_libraries(mvcad_tests PRIVATE mvcad_core Qt6::Test)
  add_test(NAME core COMMAND mvcad_tests -o ${CMAKE_BINARY_DIR}/core-results.txt,txt)
  qt_add_executable(cad_tests tests/CadTests.cpp)
  target_link_libraries(cad_tests PRIVATE mvcad_cad Qt6::Test)
  add_test(NAME cad COMMAND cad_tests -o ${CMAKE_BINARY_DIR}/cad-results.txt,txt)
  qt_add_executable(document_tests tests/DocumentTests.cpp)
  target_link_libraries(document_tests PRIVATE mvcad_document Qt6::Test)
  add_test(NAME document COMMAND document_tests -o ${CMAKE_BINARY_DIR}/document-results.txt,txt)
  qt_add_executable(document_editing_tests tests/DocumentEditingTests.cpp)
  target_link_libraries(document_editing_tests PRIVATE mvcad_document Qt6::Test)
  add_test(NAME document_editing COMMAND document_editing_tests -o ${CMAKE_BINARY_DIR}/document-editing-results.txt,txt)
  qt_add_executable(build_queue_tests tests/BuildQueueTests.cpp)
  target_link_libraries(build_queue_tests PRIVATE mvcad_builds Qt6::Test)
  add_test(NAME build_queue COMMAND build_queue_tests -o ${CMAKE_BINARY_DIR}/build-queue-results.txt,txt)
  set_tests_properties(build_queue PROPERTIES TIMEOUT 120)
  qt_add_executable(point_set_mirror_tests tests/PointSetMirrorTests.cpp)
  target_link_libraries(point_set_mirror_tests PRIVATE mvcad_document Qt6::Test)
  add_test(NAME point_set_mirror COMMAND point_set_mirror_tests -o ${CMAKE_BINARY_DIR}/point-set-mirror-results.txt,txt)
  qt_add_executable(viewport_tests tests/ViewportTests.cpp)
  target_include_directories(viewport_tests PRIVATE src)
  target_link_libraries(viewport_tests PRIVATE Qt6::Gui Qt6::Test)
  add_test(NAME viewport COMMAND viewport_tests -o ${CMAKE_BINARY_DIR}/viewport-results.txt,txt)
  add_test(NAME ui_smoke COMMAND MVCAD --smoke-test --screenshot ${CMAKE_BINARY_DIR}/ui-smoke.png)
  set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
endif()

include(GNUInstallDirs)
install(TARGETS MVCAD BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(WIN32)
  file(GLOB mvcad_occt_runtime "${OpenCASCADE_BINARY_DIR}/TK*.dll")
  install(FILES ${mvcad_occt_runtime} DESTINATION ${CMAKE_INSTALL_BINDIR})
elseif(APPLE)
  # macdeployqt copies Qt, but does not reliably discover the kernel's @rpath
  # dependencies. Include all versioned names and symlinks before deployment.
  install(DIRECTORY "${OpenCASCADE_LIBRARY_DIR}/"
    DESTINATION MVCAD.app/Contents/Frameworks
    FILES_MATCHING PATTERN "libTK*.dylib")
endif()
qt_generate_deploy_app_script(TARGET MVCAD OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
if(APPLE)
  set(mvcad_resource_dir "MVCAD.app/Contents/Resources")
else()
  set(mvcad_resource_dir ".")
endif()
install(DIRECTORY examples DESTINATION ${mvcad_resource_dir} PATTERN "extrusion-and-cut.mvcad" EXCLUDE)
install(DIRECTORY third_party DESTINATION ${mvcad_resource_dir})
install(FILES README.md DESTINATION ${mvcad_resource_dir})
install(FILES design/performance-verification-040.md DESTINATION ${mvcad_resource_dir}/design)
install(FILES docs/ROADMAP.md docs/THIRD_PARTY.md DESTINATION ${mvcad_resource_dir}/docs)
include(cmake/Packaging.cmake)
