cmake_minimum_required(VERSION 3.24)
project(MVCAD VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
qt_standard_project_setup()

add_library(mvcad_core src/core/Network.cpp src/core/PartIO.cpp src/core/Preview.cpp)
target_include_directories(mvcad_core PUBLIC src)
target_link_libraries(mvcad_core PUBLIC Qt6::Core Qt6::Gui)
if(MSVC)
  target_compile_options(mvcad_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mvcad_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(MVCAD WIN32 MACOSX_BUNDLE
  src/app/main.cpp src/app/MainWindow.cpp src/app/MainWindow.h
  src/app/Viewport.cpp src/app/Viewport.h)
target_link_libraries(MVCAD PRIVATE mvcad_core Qt6::Widgets)
target_compile_definitions(MVCAD PRIVATE MVCAD_VERSION="${PROJECT_VERSION}-dev")
if(MSVC)
  target_compile_options(MVCAD PRIVATE /W4 /permissive- /utf-8)
endif()
set_target_properties(MVCAD PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.mvcad.MVCAD
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})

if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(mvcad_tests tests/CoreTests.cpp)
  target_link_libraries(mvcad_tests PRIVATE mvcad_core Qt6::Test)
  add_test(NAME core COMMAND mvcad_tests -o ${CMAKE_BINARY_DIR}/core-results.txt,txt)
  add_test(NAME ui_smoke COMMAND MVCAD --smoke-test --screenshot ${CMAKE_BINARY_DIR}/ui-smoke.png)
  set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()

include(GNUInstallDirs)
install(TARGETS MVCAD BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
qt_generate_deploy_app_script(TARGET MVCAD OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
install(DIRECTORY examples DESTINATION .)
install(DIRECTORY third_party DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES docs/ROADMAP.md docs/THIRD_PARTY.md DESTINATION docs)
install(FILES design/command-interface-v2.md DESTINATION design)
set(CPACK_PACKAGE_NAME MVCAD)
set(CPACK_PACKAGE_VENDOR MVCAD)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Microvascular CAD - development preview")
set(CPACK_PACKAGE_INSTALL_DIRECTORY MVCAD)
set(CPACK_NSIS_DISPLAY_NAME "MVCAD Development Preview")
set(CPACK_NSIS_PACKAGE_NAME MVCAD)
set(CPACK_NSIS_EXECUTABLES_DIRECTORY bin)
set(CPACK_PACKAGE_EXECUTABLES "MVCAD;MVCAD")
set(CPACK_VERBATIM_VARIABLES YES)
if(WIN32)
  set(CPACK_GENERATOR "ZIP;NSIS")
  set(CPACK_NSIS_EXTRA_INSTALL_COMMANDS "ExecWait '\"$INSTDIR\\bin\\vc_redist.x64.exe\" /install /quiet /norestart'")
elseif(APPLE)
  set(CPACK_GENERATOR "DragNDrop")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
